        os.makedirs(output_dir, exist_ok=True)

        tz = datetime.timezone(datetime.timedelta(hours=3))
        now = datetime.datetime.now(tz)
        
        print("1. Fetching existing items from Monday to prevent duplicates...")
        monday_url = "https://api.monday.com/v2"
        monday_headers = {"Authorization": MONDAY_TOKEN, "Content-Type": "application/json", "API-Version": "2023-10"}

        query_existing = '''

                for item in items:

                    item_date = cols[0]['text'] if cols else ''
                    if item_name and item_date:

                data_rows = rows_in_page[1:]
                
            if not data_rows: break
            all_rows_content.extend(data_rows)
            if len(data_rows) < page_size: break
            page_no += 1

        print("4. Processing and sending to Monday...")
        sent_count = 0
        skipped_count = 0

        if len(all_rows_content) > 1:
            for row in all_rows_content[1:]:
                if not row or len(row) < 6: continue
                if row[5] is None: continue
                
                user_name = str(row[5]).strip()
                
                raw_date_val = row[0]
                if isinstance(raw_date_val, datetime.datetime):
                    raw_date = raw_date_val.strftime("%Y-%m-%d")
                else:
                    raw_date_str = str(raw_date_val).strip()
                    raw_date = raw_date_str.split(' ')[0]
                    if len(raw_date) < 10:

                # בדיקת הזיכרון הדינמי שלנו

        print(result_msg)
        return jsonify({"status": "success", "sent": sent_count, "skipped": skipped_count})

    except Exception as e:

        print(f"CRASHED: {error_trace}")
        return jsonify({"status": "error", "message": str(e), "trace": error_trace}), 500
